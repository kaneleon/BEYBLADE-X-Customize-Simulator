function renderPartDisplay(category, item) { if(!item) return ''; var rawCode = item.code || ''; var codeLower = rawCode.toLowerCase(); // bx-01 var codeUpper = rawCode.toUpperCase(); // BX-01 var codeNoHyphenLower = codeLower.replace(/-/g, ''); // bx01 var codeNoHyphenUpper = codeUpper.replace(/-/g, ''); // BX01 var nameFull = item.name; // 例: ドランソード var nameFirstWord = item.name.split(' ')[0]; // 例: F (Bit用) var baseNames = []; if (category === 'blade') { baseNames = [ codeNoHyphenLower, // bx01 codeNoHyphenUpper, // BX01 codeLower, // bx-01 codeUpper, // BX-01 codeNoHyphenLower + nameFull, // bx01ドランソード codeNoHyphenUpper + nameFull, // BX01ドランソード nameFull // ドランソード ]; } else if (category === 'ratchet') { baseNames = [ nameFull, // 3-60 nameFull.replace(/-/g, '') // 360 ]; } else if (category === 'bit') { baseNames = [ nameFirstWord, // F nameFirstWord.toLowerCase(), // f nameFirstWord.toUpperCase() // F ]; } // 小文字・大文字拡張子の両方を網羅 var exts = ['.png', '.PNG', '.jpg', '.JPG', '.jpeg', '.JPEG', '.webp', '.WEBP']; var candidates = []; for (var i = 0; i < baseNames.length; i++) { for (var j = 0; j < exts.length; j++) { var path = 'images/' + encodeURIComponent(baseNames[i]) + exts[j]; if (candidates.indexOf(path) === -1) { candidates.push(path); } } } var candidatesStr = candidates.join('|'); return ''; }
