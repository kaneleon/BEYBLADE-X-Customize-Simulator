Number(pStats.totalWeight) : pStats[targetKey]; combos.push({ bIdx: i, rIdx: j, btIdx: k, blade: bd, ratchet: rt, bit: bt, score: score, weight: Number(pStats.totalWeight) }); } } } combos.sort(function(a, b) { return (b.score - a.score) || (b.weight - a.weight); }); var topCombos = []; var usedBlades = {}; for (var c = 0; c < combos.length; c++) { var item = combos[c]; if(!usedBlades[item.blade.name]) { topCombos.push(item); usedBlades[item.blade.name] = true; } if(topCombos.length >= 6) break; } // 計算結果をキャッシュに保存 recCache[targetKey] = topCombos; this.renderRecommendations(topCombos, targetKey); } catch(err) { console.error("Recommend error:", err); } }, // 描画ロジックの分離関数 renderRecommendations: function(topCombos, targetKey) { var labels = { atk: '攻撃力', def: '防御力', sta: '持久力', dash: 'ダッシュ力', brst: 'バースト耐性', weight: '総重量' }; var grid = document.getElementById('recommendGrid'); if(!grid) return; grid.innerHTML = ''; topCombos.forEach(function(c, rank) { var scoreText = targetKey === 'weight' ? c.weight.toFixed(2) + ' g' : c.score.toFixed(1) + ' pt'; var card = document.createElement('div'); card.className = 'recommend-card'; var rankBadge = document.createElement('span'); rankBadge.className = 'rank-badge'; rankBadge.textContent = 'RANK ' + (rank + 1); var title = document.createElement('h3'); title.textContent = '[' + c.blade.code + '] ' + c.blade.name + ' ' + c.ratchet.name + ' ' + c.bit.name.split(' ')[0]; var detail = document.createElement('div'); detail.className = 'combo-detail'; detail.innerHTML = 'B: [' + c.blade.code + '] ' + c.blade.name + '
R: ' + c.ratchet.name + '
